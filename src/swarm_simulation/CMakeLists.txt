cmake_minimum_required(VERSION 3.0.2)
project(swarm_simulation)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs gazebo_ros)
find_package(gazebo REQUIRED)
catkin_package()
include_directories(include ${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${GAZEBO_CXX_FLAGS}")
add_library(swarm_camera_imu SHARED src/gazebo_ros_imu_sensor.cpp)
target_link_libraries(swarm_camera_imu ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
install(TARGETS swarm_camera_imu LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY models worlds launch scripts DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
