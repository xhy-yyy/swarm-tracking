cmake_minimum_required(VERSION 3.0.2)
project(actor_collisions)
find_package(catkin REQUIRED COMPONENTS gazebo_ros)
find_package(gazebo REQUIRED)
catkin_package()
include_directories(${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${GAZEBO_CXX_FLAGS}")
add_library(ActorCollisionsPlugin SHARED ActorCollisionsPlugin.cc)
target_link_libraries(ActorCollisionsPlugin ${GAZEBO_LIBRARIES})
install(TARGETS ActorCollisionsPlugin LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES actor_collisions.world DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
